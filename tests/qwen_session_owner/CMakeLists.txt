if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT TARGET strix_serve)
    project(strix_qwen_session_owner_host LANGUAGES CXX)

    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../session_runtime" shared)

get_filename_component(QWEN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT TARGET strix_qwen_http_session_owner)
    add_library(strix_qwen_http_session_owner STATIC "${QWEN_ROOT}/src/qwen/http_session_owner.cpp")
    target_include_directories(strix_qwen_http_session_owner PUBLIC "${QWEN_ROOT}/src")

    strix_session_warnings(strix_qwen_http_session_owner)
endif()

# Compile the actual production binding against the host ABI. Fail closed on
# either changed/duplicate marker. This is not a copied owner algorithm.
file(READ "${QWEN_ROOT}/apps/hip_http_serve.hpp" source)
set(begin "        // Qwen native-control binding begin.")
set(end "        // Qwen native-control binding end.")
foreach(marker IN ITEMS "${begin}" "${end}")
    string(FIND "${source}" "${marker}" pos)
    if(pos LESS 0)
        message(FATAL_ERROR "Missing native binding marker")
    endif()
    string(LENGTH "${marker}" len)
    math(EXPR tail "${pos}+${len}")
    string(SUBSTRING "${source}" ${tail} -1 rest)
    string(FIND "${rest}" "${marker}" duplicate)
    if(NOT duplicate LESS 0)
        message(FATAL_ERROR "Duplicate native binding marker")
    endif()
endforeach()
string(FIND "${source}" "${begin}" start)
string(FIND "${source}" "${end}" finish)
math(EXPR length "${finish}-${start}")
if(length LESS 1)
    message(FATAL_ERROR "Reversed native binding markers")
endif()
string(SUBSTRING "${source}" ${start} ${length} binding)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/current_binding.inc" "${binding}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${QWEN_ROOT}/apps/hip_http_serve.hpp")

add_executable(qwen_session_owner_tests replay_tests.cpp

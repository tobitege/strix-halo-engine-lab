cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_http_shutdown_host LANGUAGES CXX)
    include(CTest)
    find_package(Threads REQUIRED)
    get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    add_library(strix_http_shutdown_server STATIC "${ROOT}/src/serve/http_server.cpp")
    target_include_directories(strix_http_shutdown_server PUBLIC "${ROOT}/include")
    target_compile_features(strix_http_shutdown_server PUBLIC cxx_std_20)
    target_link_libraries(strix_http_shutdown_server PUBLIC Threads::Threads)
    if(WIN32)
        target_compile_definitions(strix_http_shutdown_server PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(strix_http_shutdown_server PUBLIC ws2_32)
    endif()
    set(HTTP_SHUTDOWN_LIBRARY strix_http_shutdown_server)
endif()
if(NOT HTTP_SHUTDOWN_LIBRARY)
    set(HTTP_SHUTDOWN_LIBRARY strix_serve)
endif()
add_executable(http_shutdown_tests shutdown_tests.cpp)
target_link_libraries(http_shutdown_tests PRIVATE ${HTTP_SHUTDOWN_LIBRARY})
if(MSVC)
    target_compile_options(http_shutdown_tests PRIVATE /EHsc /utf-8 /W4)
endif()
foreach(case idle keepalive header body write accept repeat refusal stop-refusal bytes sse deadline)
    add_test(NAME http-shutdown-${case} COMMAND http_shutdown_tests ${case})
    set_tests_properties(http-shutdown-${case} PROPERTIES TIMEOUT 8 LABELS "cpu;http;shutdown")
endforeach()
# Require both the exact exit code and exact diagnostic, including Release.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sentinel.cmake" [=[
execute_process(COMMAND "${BINARY}" --check-sentinel RESULT_VARIABLE rc OUTPUT_VARIABLE out ERROR_VARIABLE err TIMEOUT 5)
if(NOT rc EQUAL 1 OR NOT out STREQUAL "" OR NOT err STREQUAL "http shutdown check: false\n")
    message(FATAL_ERROR "shutdown negative sentinel mismatch: ${rc}: ${out}${err}")
endif()
]=])
add_test(NAME http-shutdown-checks COMMAND "${CMAKE_COMMAND}" "-DBINARY=$<TARGET_FILE:http_shutdown_tests>"
    -P "${CMAKE_CURRENT_BINARY_DIR}/sentinel.cmake")
set_tests_properties(http-shutdown-checks PROPERTIES TIMEOUT 8)

cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_ds41_dense_forms_host LANGUAGES CXX)
    enable_testing()
endif()
# Host checks of the DS41 dense-forms kit: bounded --forms arguments and the wrapper's abort-on-failure contract.
add_executable(strix-ds41-dense-forms-args-tests args.cpp)
target_compile_features(strix-ds41-dense-forms-args-tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-ds41-dense-forms-args-tests PRIVATE /W4 /WX)
else()
    target_compile_options(strix-ds41-dense-forms-args-tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME strix-ds41-dense-forms-args-tests COMMAND strix-ds41-dense-forms-args-tests)
find_package(Python3 COMPONENTS Interpreter)
find_program(STRIX_DENSE_FORMS_BASH bash)
if(Python3_Interpreter_FOUND AND STRIX_DENSE_FORMS_BASH AND NOT WIN32)
    add_test(NAME strix-ds41-dense-forms-wrapper-tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/wrapper.py")
    set_tests_properties(strix-ds41-dense-forms-wrapper-tests PROPERTIES ENVIRONMENT "DENSE_FORMS_BASH=${STRIX_DENSE_FORMS_BASH}")
endif()

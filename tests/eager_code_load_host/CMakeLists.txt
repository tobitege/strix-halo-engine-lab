cmake_minimum_required(VERSION 3.20)
project(eager_code_load_host LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(strix-eager-code-load-host eager_restart_tests.cpp)
target_include_directories(strix-eager-code-load-host PRIVATE ../../include)
target_compile_features(strix-eager-code-load-host PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(strix-eager-code-load-host PRIVATE /W4 /WX /permissive-)
    target_compile_definitions(strix-eager-code-load-host PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(strix-eager-code-load-host PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME strix-eager-code-load-restart-failure COMMAND strix-eager-code-load-host failure)
add_test(NAME strix-eager-code-load-explicit-hip COMMAND strix-eager-code-load-host explicit)
add_test(NAME strix-eager-code-load-kfd-reports COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/kfdevict_tests.py")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    foreach(case deleted replaced denied)
        add_test(NAME strix-eager-code-load-restart-${case} COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/linux_restart_test.py" "$<TARGET_FILE:strix-eager-code-load-host>"
            "${CMAKE_CURRENT_BINARY_DIR}" "${case}")
    endforeach()
endif()

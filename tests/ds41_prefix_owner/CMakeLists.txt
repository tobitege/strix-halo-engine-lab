cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_ds41_prefix_owner LANGUAGES CXX)
    include(CTest)
    get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    include("${REPO_ROOT}/cmake/StrixStateCache.cmake")
else()
    set(REPO_ROOT "${PROJECT_SOURCE_DIR}")
endif()
# Actual family source with the minimal host-only production owner/IO closure.
add_library(ds41_prefix_owner_fixture STATIC
    "${REPO_ROOT}/src/deepseek41_prefix_cache.cpp"
    "${REPO_ROOT}/src/deepseek41_attention_state.cpp")
target_link_libraries(ds41_prefix_owner_fixture PUBLIC strix_state_cache)
if(MSVC)
    target_compile_options(ds41_prefix_owner_fixture PRIVATE /EHsc /utf-8 /W4)
else()
    target_compile_options(ds41_prefix_owner_fixture PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(ds41_prefix_owner_tests owner_tests.cpp)
target_link_libraries(ds41_prefix_owner_tests PRIVATE ds41_prefix_owner_fixture)
target_compile_features(ds41_prefix_owner_tests PRIVATE cxx_std_20)
target_compile_definitions(ds41_prefix_owner_tests PRIVATE STRIX_DS41_OWNER_ADOPTED)
if(MSVC)
    target_compile_options(ds41_prefix_owner_tests PRIVATE /EHsc /utf-8 /W4)
endif()
add_test(NAME ds41_prefix_owner COMMAND ds41_prefix_owner_tests)
set_tests_properties(ds41_prefix_owner PROPERTIES TIMEOUT 90 LABELS "cpu;session;prefix")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sentinel-$<CONFIG>.cmake" CONTENT
"execute_process(COMMAND \"$<TARGET_FILE:ds41_prefix_owner_tests>\" --fail-sentinel RESULT_VARIABLE rc OUTPUT_VARIABLE out ERROR_VARIABLE err TIMEOUT 10)
if(NOT rc EQUAL 1 OR NOT out STREQUAL \"\" OR NOT err STREQUAL \"DS41 prefix owner FAIL: deliberate active-check sentinel\\n\")
 message(FATAL_ERROR \"sentinel rc=\${rc} out=\${out} err=\${err}\")
endif()
")
add_test(NAME ds41_prefix_owner_checks COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/sentinel-$<CONFIG>.cmake")
set_tests_properties(ds41_prefix_owner_checks PROPERTIES TIMEOUT 15 LABELS "cpu;session;prefix")

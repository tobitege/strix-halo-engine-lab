cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(strix_ds41_owner_host LANGUAGES CXX)
    include(CTest)
    if(NOT DEFINED DS41_OWNER_ROOT)
        get_filename_component(DS41_OWNER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    endif()
    add_subdirectory("${DS41_OWNER_ROOT}/tests/session_runtime" shared)
endif()
if(NOT DEFINED DS41_OWNER_ROOT)
    set(DS41_OWNER_ROOT "${PROJECT_SOURCE_DIR}")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(DS41_OWNER_ENTRY "Build an independent archived pre-adoption loop" OFF)
set(DS41_OWNER_SOURCE "${DS41_OWNER_ROOT}/apps/deepseek41_serve_main.cpp" CACHE FILEPATH "Actual lead ownership source")
execute_process(COMMAND "${CMAKE_COMMAND}" "-DPYTHON=${Python3_EXECUTABLE}" "-DSOURCE=${DS41_OWNER_SOURCE}"
    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/extracted" -P "${CMAKE_CURRENT_LIST_DIR}/extract_reference.cmake" RESULT_VARIABLE rc)
if(NOT rc EQUAL 0)
    message(FATAL_ERROR "DS41 actual lead extraction refused")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DS41_OWNER_SOURCE}" "${CMAKE_CURRENT_LIST_DIR}/extract_reference.cmake")
add_executable(ds41_session_owner_tests owner_tests.cpp)
if(TARGET strix_ds41_session)
    target_link_libraries(ds41_session_owner_tests PRIVATE strix_ds41_session)
else()
    # Standalone fixture: the actual family codec/state/compatibility sources.
    # Queue, HTTP, Runtime, logical leases, cache and IO use the common closure.
    target_sources(ds41_session_owner_tests PRIVATE
        "${DS41_OWNER_ROOT}/src/deepseek41_session_cache.cpp"
        "${DS41_OWNER_ROOT}/src/deepseek41_attention_state.cpp"
        "${DS41_OWNER_ROOT}/src/ds41_native_fp8.cpp"
        "${DS41_OWNER_ROOT}/src/ds41_trellis.cpp"
        "${DS41_OWNER_ROOT}/src/ds41_attention.cpp"
        "${DS41_OWNER_ROOT}/src/ds41_formats.cpp"
        "${DS41_OWNER_ROOT}/src/gguf_reader.cpp")
endif()
target_include_directories(ds41_session_owner_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/extracted")
target_link_libraries(ds41_session_owner_tests PRIVATE strix_serve strix_session_owner)
if(NOT DS41_OWNER_ENTRY)
    target_compile_definitions(ds41_session_owner_tests PRIVATE DS41_OWNER_ADOPTED=1)
    if(NOT TARGET strix_ds41_http_session_owner)
        add_library(strix_ds41_http_session_owner STATIC "${DS41_OWNER_ROOT}/src/deepseek41/http_session_owner.cpp")
        target_include_directories(strix_ds41_http_session_owner PUBLIC "${DS41_OWNER_ROOT}/src")
        target_link_libraries(strix_ds41_http_session_owner PUBLIC strix_session_owner)
        strix_session_warnings(strix_ds41_http_session_owner)
    endif()
    target_link_libraries(ds41_session_owner_tests PRIVATE strix_ds41_http_session_owner)
endif()
target_compile_features(ds41_session_owner_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(ds41_session_owner_tests PRIVATE /EHsc /utf-8 /W4 /Gy)
    target_link_options(ds41_session_owner_tests PRIVATE /OPT:REF)
endif()
add_test(NAME ds41_session_owner COMMAND ds41_session_owner_tests)
add_test(NAME ds41_session_owner_extended COMMAND ds41_session_owner_tests --extended)
if(NOT DS41_OWNER_ENTRY)
    add_test(NAME ds41_session_owner_failure COMMAND ds41_session_owner_tests --failure)
    set_tests_properties(ds41_session_owner_failure PROPERTIES TIMEOUT 30 LABELS "cpu;session;ds41-owner")
endif()
add_test(NAME ds41_session_owner_checks COMMAND "${CMAKE_COMMAND}" "-DBINARY=$<TARGET_FILE:ds41_session_owner_tests>"
    -DCHECK_FAILURE=ON -P "${CMAKE_CURRENT_LIST_DIR}/extract_reference.cmake")
set_tests_properties(ds41_session_owner ds41_session_owner_extended ds41_session_owner_checks PROPERTIES TIMEOUT 30 LABELS "cpu;session;ds41-owner")
set(DS41_OWNER_REFERENCE "" CACHE FILEPATH "Immutable actual entry lead capture")
if(DS41_OWNER_REFERENCE)
    file(SHA256 "${DS41_OWNER_REFERENCE}" reference_sha)
    if(NOT reference_sha STREQUAL "d2ffc83a201e05d49062f9df67d9fdeedba3317bda3282d4e1e4b4e9f3764a3a")
        message(FATAL_ERROR "Unexpected immutable DS41 original capture")
    endif()
    add_test(NAME ds41_session_owner_reference COMMAND ds41_session_owner_tests --compare "${DS41_OWNER_REFERENCE}")
    set_tests_properties(ds41_session_owner_reference PROPERTIES TIMEOUT 30 LABELS "cpu;session;ds41-owner;reference")
endif()
set(DS41_OWNER_EXTENDED_REFERENCE "" CACHE FILEPATH "Separate supplemental entry lead capture")
if(DS41_OWNER_EXTENDED_REFERENCE)
    file(SHA256 "${DS41_OWNER_EXTENDED_REFERENCE}" reference_sha)
    if(NOT reference_sha STREQUAL "debd787760cfd4460fc660d667e6a2d62aaae3791da4f2e1bd2603a47d095765")
        message(FATAL_ERROR "Unexpected separate supplemental DS41 capture")
    endif()
    add_test(NAME ds41_session_owner_extended_reference COMMAND ds41_session_owner_tests --compare-extended "${DS41_OWNER_EXTENDED_REFERENCE}")
    set_tests_properties(ds41_session_owner_extended_reference PROPERTIES TIMEOUT 30 LABELS "cpu;session;ds41-owner;reference")
endif()

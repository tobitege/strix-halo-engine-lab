cmake_minimum_required(VERSION 3.20)
project(strix_session_backend_tests LANGUAGES CXX)
include(CTest)
get_filename_component(STRIX_REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(strix_session_backend_cpu STATIC "${STRIX_REPOSITORY_ROOT}/src/session/backend_cpu.cpp")
target_include_directories(strix_session_backend_cpu PUBLIC "${STRIX_REPOSITORY_ROOT}/include")
target_compile_features(strix_session_backend_cpu PUBLIC cxx_std_20)
set_target_properties(strix_session_backend_cpu PROPERTIES CXX_EXTENSIONS OFF)
if(BUILD_TESTING)
    add_executable(session_backend_tests "${STRIX_REPOSITORY_ROOT}/tests/session_backend_tests.cpp")
    target_link_libraries(session_backend_tests PRIVATE strix_session_backend_cpu)
    foreach(target strix_session_backend_cpu session_backend_tests)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        endif()
    endforeach()
    add_test(NAME session_backend COMMAND session_backend_tests)
endif()

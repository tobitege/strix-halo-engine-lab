cmake_minimum_required(VERSION 3.24)
project(wfmt_early_registration LANGUAGES CXX)
enable_testing()
# Reproduce the root's early subdirectory entry, without configuring an engine.
get_filename_component(STRIX_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${STRIX_ROOT}/tests/wfmt" wfmt)
include(CTest)
include("${STRIX_ROOT}/cmake/StrixBackendCpu.cmake")
include("${STRIX_ROOT}/cmake/StrixIntel.cmake")
foreach(target strix-cpu-backend-host-tests strix-intel-backend-host-tests strix-intel-sycl-boundary-tests)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Fresh early wfmt configure omitted ${target}")
    endif()
endforeach()
get_property(registered DIRECTORY "${STRIX_ROOT}/tests/wfmt" PROPERTY TESTS)
foreach(test strix-cpu-backend-host-tests strix-intel-sycl-async-failures strix-intel-sycl-sync-failures strix-intel-sycl-open-failures)
    if(NOT test IN_LIST registered)
        message(FATAL_ERROR "Fresh early wfmt configure omitted test ${test}")
    endif()
endforeach()

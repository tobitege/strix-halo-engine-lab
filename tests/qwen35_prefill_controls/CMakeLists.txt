cmake_minimum_required(VERSION 3.24)
project(qwen35_prefill_controls LANGUAGES CXX)
include(CTest)
add_executable(qwen35-prefill-controls ../qwen35_prefill_controls_tests.cpp)
target_include_directories(qwen35-prefill-controls PRIVATE ../../include)
target_compile_features(qwen35-prefill-controls PRIVATE cxx_std_20)
if(MSVC)
 target_compile_options(qwen35-prefill-controls PRIVATE /W4 /WX /utf-8)
else()
 target_compile_options(qwen35-prefill-controls PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME qwen35-prefill-controls COMMAND qwen35-prefill-controls)

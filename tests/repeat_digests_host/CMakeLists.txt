# Compile the production timing loop and digest adapter with host fakes, not a
# reimplementation. Fail configuration if these function boundaries change.
set(app "${STRIX_REPEAT_ROOT}/apps/deepseek41_prefill_main.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${app}")
file(READ "${app}" source)
function(extract begin end output)
    string(FIND "${source}" "${begin}" first)
    string(FIND "${source}" "${end}" last)
    if(first LESS 0 OR last LESS_EQUAL first)
        message(FATAL_ERROR "Production timing test extraction boundary missing: ${begin}")
    endif()
    math(EXPR length "${last} - ${first}")
    string(SUBSTRING "${source}" ${first} ${length} body)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}")
endfunction()
extract("strix::RepeatDigests::Run repeat_run(" "void print_result(" repeat_adapter.inc)
extract("strix::deepseek41::PrefillResult ttft_runs(" "// --plan:" ttft_runs.inc)
add_executable(strix-ttft-repeat-tests ttft_repeat_tests.cpp)
target_include_directories(strix-ttft-repeat-tests PRIVATE "${STRIX_REPEAT_ROOT}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(strix-ttft-repeat-tests PRIVATE cxx_std_20)
foreach(target strix-repeat-digests-tests strix-ttft-repeat-tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()

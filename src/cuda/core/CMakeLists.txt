get_filename_component(STRIX_NVFP4_ROOT ../../.. ABSOLUTE)
option(STRIX_NVFP4_CUDA "Compile native SM120 NVFP4 operator" OFF)
add_library(strix_nvfp4_reference STATIC nvfp4_reference.cpp)
target_include_directories(strix_nvfp4_reference PUBLIC ${STRIX_NVFP4_ROOT}/include)
target_compile_features(strix_nvfp4_reference PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(strix_nvfp4_reference PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(strix_nvfp4_reference PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(STRIX_NVFP4_CUDA)
  enable_language(CUDA)
  add_library(strix_nvfp4 STATIC nvfp4.cu)
  target_link_libraries(strix_nvfp4 PUBLIC strix_nvfp4_reference)
  set_target_properties(strix_nvfp4 PROPERTIES CUDA_ARCHITECTURES 120a CUDA_STANDARD 17)
  target_compile_options(strix_nvfp4 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()

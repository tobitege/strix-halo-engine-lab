# Standalone configuration of the Intel backend scaffold and its host tests (no HIP, no CUDA):
#   cmake -S src/intel -B build-intel && cmake --build build-intel --config Release && ctest --test-dir build-intel -C Release
# The full build reaches the same targets with -DSTRIX_BACKEND_INTEL=ON.
cmake_minimum_required(VERSION 3.24)
project(strix_intel_backend LANGUAGES CXX)
include(CTest)
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    add_compile_options(/Zc:__cplusplus /permissive-)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()
function(strix_target_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
endfunction()
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/StrixIntel.cmake)
option(STRIX_WFMT_TESTS "Build weight-format qualification targets" OFF)
if(STRIX_WFMT_TESTS)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../tests/wfmt ${CMAKE_CURRENT_BINARY_DIR}/wfmt)
endif()
